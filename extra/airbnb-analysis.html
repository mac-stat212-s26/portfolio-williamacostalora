Introduction

This project uses NYC Airbnb data to demonstrate four key concepts we have learned these past couple of weeks in class:

  1. Missing Data: Finding and handling explicit/implicit missing values
  2. Custom Functions: Writing reusable functions with tidy evaluation
  3. Base R Subsetting: Using $, [[]], and []
  4. Iteration: For loops and across() for repeated tasks
Code
library(tidyverse)
library(knitr)

airbnb <- read_csv("data/raw/archive/AB_NYC_2019.csv") |> 
  mutate(last_review = mdy(last_review))

Concept 1: Missing Data

Finding Explicit Missing Values

Code
# Count missing values in each column
airbnb |> 
  summarize(
    name = sum(is.na(name)),
    host_name = sum(is.na(host_name)),
    last_review = sum(is.na(last_review)),
    reviews_per_month = sum(is.na(reviews_per_month))
  ) |> 
  kable(caption = "Count of Missing Values")

Observation: last_review and reviews_per_month have 10,052 missing values - these are listings that haven’t been reviewed yet.

Handling Missing Values

Code
# Fill missing reviews_per_month with 0
airbnb_clean <- airbnb |> 
  mutate(reviews_per_month = coalesce(reviews_per_month, 0))

# Verify
tibble(
  before = sum(is.na(airbnb$reviews_per_month)),
  after = sum(is.na(airbnb_clean$reviews_per_month))
) |> 
  kable(caption = "Missing values before and after coalesce()")

# Function to calculate what % of listings meet a condition
calc_proportion <- function(data, condition) {
  data |> 
    summarize(
      n_total = n(),
      n_meeting_condition = sum({{ condition }}, na.rm = TRUE),
      proportion = mean({{ condition }}, na.rm = TRUE)
    )
}

# Function with embracing and optional grouping
price_summary <- function(data, group_var) {
  data |> 
    group_by({{ group_var }}) |>

Concept 3: Base R Subsetting

Using $, [[]], and []

Code
# Method 1: Extract column with $
brooklyn_prices_dollar <- airbnb_clean$price[airbnb_clean$neighbourhood_group == "Brooklyn"]

# Method 2: Extract column with [[]]
brooklyn_data <- airbnb_clean[airbnb_clean$neighbourhood_group == "Brooklyn", ]
brooklyn_prices_bracket <- brooklyn_data[["price"]]

# Method 3: Tidyverse (for comparison)
brooklyn_prices_tidy <- airbnb_clean |> 
  filter(neighbourhood_group == "Brooklyn") |> 
  pull(price)

# Verify all three methods give same result
tibble(
  method = c("$ operator", "[[ ]] operator", "tidyverse"),
  n_values = c(length(brooklyn_prices_dollar), 
               length(brooklyn_prices_bracket), 
               length(brooklyn_prices_tidy)),
  mean_price = c(mean(brooklyn_prices_dollar), 
                 mean(brooklyn_prices_bracket), 
                 mean(brooklyn_prices_tidy)),
  identical = c(
    identical(brooklyn_prices_dollar, brooklyn_prices_bracket),
    identical(brooklyn_prices_bracket, brooklyn_prices_tidy),
    TRUE
  )
) |> 
  kable(digits = 2, caption = "Comparison of Subsetting Methods")
Comparison of Subsetting Methods
method n_values mean_price identical

Concept 4: Iteration

For Loop with Pre-allocation

Code
# Goal: Calculate median price for each borough

# Get unique boroughs
boroughs <- unique(airbnb_clean$neighbourhood_group)

# Pre-allocate storage
median_prices <- vector("numeric", length = length(boroughs))
names(median_prices) <- boroughs

# For loop with seq_along
for (i in seq_along(boroughs)) {
  borough_name <- boroughs[i]
  
  # Subset data
  borough_data <- airbnb_clean |> 
    filter(neighbourhood_group == borough_name)
  
  # Calculate and store
  median_prices[i] <- median(borough_data$price)
}

# Display results
data.frame(
  borough = names(median_prices),
  median_price = median_prices
) |> 
  arrange(desc(median_price)) |> 
  kable(caption = "Median Price by Borough (for loop)")

Code
# Function to analyze one scenario
analyze_scenario <- function(data, borough, expensive) {
  filtered <- data |> 
    filter(neighbourhood_group == borough)
  
  if (expensive) {
    filtered <- filtered |> filter(price > 150)
  }
  
  tibble(
    n_listings = nrow(filtered),
    avg_price = mean(filtered$price)
  )
}

# Run all scenarios with for loop
results <- vector("list", length = nrow(scenarios))

for (i in seq_len(nrow(scenarios))) {
  results[[i]] <- analyze_scenario(
    data = airbnb_clean,
    borough = scenarios$borough[i],
    expensive = scenarios$expensive[i]
  )
}

# Combine results
bind_cols(scenarios, bind_rows(results)) |> 
  kable(digits = 2, caption = "Results for All Scenarios")
Results for All Scenarios
